// ============================================================================ // CONEXIÓN DIRECTA A GOOGLE SHEETS // ============================================================================ const SPREADSHEET_ID = "1vhUX9wtMLGWXhxJXbbUOHGUT6YwjLpSAfuLFBLK2cVc"; function getSpreadsheet() { return SpreadsheetApp.openById(SPREADSHEET_ID); } // Inicializa las 5 pestañas esenciales function setupSheets() { const ss = getSpreadsheet(); const sheets = ['Clientes', 'Productos', 'Ventas', 'Seguimiento', 'Caja']; sheets.forEach(name => { if (!ss.getSheetByName(name)) { ss.insertSheet(name); } }); const shClientes = ss.getSheetByName('Clientes'); if (shClientes.getLastRow() === 0) shClientes.appendRow(['ID', 'Nombre', 'Teléfono', 'Email', 'Dirección']); const shProductos = ss.getSheetByName('Productos'); if (shProductos.getLastRow() === 0) shProductos.appendRow(['ID', 'Nombre', 'Precio', 'Stock', 'Imagen']); const shVentas = ss.getSheetByName('Ventas'); if (shVentas.getLastRow() === 0) shVentas.appendRow(['ID Ventas', 'Fecha', 'Cliente', 'Total', 'Detalle JSON']); const shSeguimiento = ss.getSheetByName('Seguimiento'); if (shSeguimiento.getLastRow() === 0) shSeguimiento.appendRow(['ID', 'Cliente', 'Monto', 'Estado', 'Fecha Recordatorio', 'Notas']); const shCaja = ss.getSheetByName('Caja'); if (shCaja.getLastRow() === 0) shCaja.appendRow(['Fecha', 'Tipo', 'Monto', 'Concepto']); } // ============================================================================ // RECEPTOR DE SOLICITUDES (GET Y POST) // ============================================================================ function doGet(e) { let response = {}; try { const action = e.parameter ? e.parameter.action : ''; if (action === 'getData') { response = { status: 'success', clientes: getTableData('Clientes'), productos: getTableData('Productos'), seguimiento: getTableData('Seguimiento'), balance: getBalanceData() }; } else { response = { status: 'error', message: 'Acción GET requerida' }; } } catch (err) { response = { status: 'error', message: err.toString() }; } return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { let response = { status: 'success' }; try { let action, payload; // Soporte para JSON enviado por fetch if (e.postData && e.postData.contents) { let body = JSON.parse(e.postData.contents); action = body.action; payload = body.payload; } else { action = e.parameter.action; payload = JSON.parse(e.parameter.payload || '{}'); } if (action === 'guardarProducto') { guardarProducto(payload); } else if (action === 'guardarCliente') { guardarCliente(payload); } else if (action === 'registrarVenta') { procesarVenta(payload); } else if (action === 'actualizarEstadoPago') { actualizarEstadoSeguimiento(payload.id, payload.nuevoEstado); } } catch (err) { response = { status: 'error', message: err.toString() }; } return ContentService.createTextOutput(JSON.stringify(response)) .setMimeType(ContentService.MimeType.JSON); } // ============================================================================ // OPERACIONES EN LA HOJA // ============================================================================ function getTableData(sheetName) { const sheet = getSpreadsheet().getSheetByName(sheetName); if (!sheet) return []; const data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; const headers = data[0]; return data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => { let key = h.toString().toLowerCase().split(' ')[0].replace(/[^a-z0-9]/g, ''); obj[key] = row[i]; }); return obj; }); } function guardarProducto(p) { const sheet = getSpreadsheet().getSheetByName('Productos'); sheet.appendRow([ p.id || 'PROD-' + Date.now(), p.nombre, p.precio, p.stock, p.imagen || '' ]); } function guardarCliente(c) { const sheet = getSpreadsheet().getSheetByName('Clientes'); sheet.appendRow([ c.id || 'CLI-' + Date.now(), c.nombre, c.telefono || '', c.email || '', c.direccion || 'Guatemala' ]); } function procesarVenta(venta) { const ss = getSpreadsheet(); const shVentas = ss.getSheetByName('Ventas'); const shProductos = ss.getSheetByName('Productos'); const shSeguimiento = ss.getSheetByName('Seguimiento'); const shCaja = ss.getSheetByName('Caja'); const idVenta = 'V-' + new Date().getTime(); const fecha = new Date().toISOString().split('T')[0]; shVentas.appendRow([ idVenta, fecha, venta.cliente || 'Cliente General', venta.total, JSON.stringify(venta.items) ]); const prodData = shProductos.getDataRange().getValues(); venta.items.forEach(item => { for (let i = 1; i < prodData.length; i++) { if (prodData[i][0] == item.id || prodData[i][1] == item.nombre) { let stockActual = Number(prodData[i][3]) || 0; let nuevoStock = Math.max(0, stockActual - item.cantidad); shProductos.getRange(i + 1, 4).setValue(nuevoStock); break; } } }); if (venta.estadoPago === 'pendiente') { shSeguimiento.appendRow([ 'SEG-' + Date.now(), venta.cliente || 'Cliente General', venta.total, 'pendiente', fecha, 'Venta a crédito ID: ' + idVenta ]); } else { shCaja.appendRow([ fecha, 'Ingreso', venta.total, 'Venta al contado ID: ' + idVenta ]); } } function actualizarEstadoSeguimiento(idSeguimiento, nuevoEstado) { const ss = getSpreadsheet(); const shSeg = ss.getSheetByName('Seguimiento'); const data = shSeg.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] == idSeguimiento) { shSeg.getRange(i + 1, 4).setValue(nuevoEstado); if (nuevoEstado === 'pagado') { const cliente = data[i][1]; const monto = data[i][2]; const fechaActual = new Date().toISOString().split('T')[0]; ss.getSheetByName('Caja').appendRow([ fechaActual, 'Ingreso', monto, 'Cobro de cuenta pendiente: ' + cliente ]); } break; } } } function getBalanceData() { const shCaja = getSpreadsheet().getSheetByName('Caja'); if (!shCaja) return { ingresos: 0, egresos: 0, balanceTotal: 0 }; const data = shCaja.getDataRange().getValues(); let totalIngresos = 0; let totalEgresos = 0; for (let i = 1; i < data.length; i++) { let tipo = data[i][1]; let monto = Number(data[i][2]) || 0; if (tipo === 'Ingreso') totalIngresos += monto; if (tipo === 'Egreso') totalEgresos += monto; } return { ingresos: totalIngresos, egresos: totalEgresos, balanceTotal: totalIngresos - totalEgresos }; }
